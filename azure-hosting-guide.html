How the System Works on Azure

The portal uses a simple two-tier Azure architecture: static HTML files served from Azure Blob Storage, and a lightweight Azure Function App that handles token validation, creation, and submission storage. No servers to manage.

Function 1 of 3

validate-token — Checks if a link is valid, not expired, and not used

File: ae-functions/validate-token/index.js
const { BlobServiceClient } = require("@azure/storage-blob");

module.exports = async function(context, req) {
  const token = req.query.token;
  if (!token) {
    context.res = { status: 400, body: { valid: false, reason: "No token provided" }};
    return;
  }

  try {
    const client = BlobServiceClient.fromConnectionString(process.env.STORAGE_CONNECTION);
    const container = client.getContainerClient("tokens");
    const blob = container.getBlobClient(token + ".json");
    
    const download = await blob.download();
    const text = await streamToText(download.readableStreamBody);
    const data = JSON.parse(text);
    
    // Check expiry
    if (new Date(data.expiresAt) < new Date()) {
      context.res = { body: { valid: false, reason: "This assessment link has expired." }};
      return;
    }
    // Check already used
    if (data.status === "submitted") {
      context.res = { body: { valid: false, reason: "This link has already been submitted." }};
      return;
    }
    
    context.res = { body: { valid: true, assessmentType: data.type, company: data.company }};
  } catch(e) {
    context.res = { body: { valid: false, reason: "Invalid or unrecognised link." }};
  }
};

async function streamToText(stream) {
  const chunks = [];
  for await (const chunk of stream) chunks.push(chunk);
  return Buffer.concat(chunks).toString("utf-8");
}

Function 2 of 3

create-token — Admin creates a new assessment link (called from admin.html)

File: ae-functions/create-token/index.js
const { BlobServiceClient } = require("@azure/storage-blob");

module.exports = async function(context, req) {
  const body = req.body;
  if (!body || !body.token) {
    context.res = { status: 400, body: { success: false }};
    return;
  }

  try {
    const client = BlobServiceClient.fromConnectionString(process.env.STORAGE_CONNECTION);
    const container = client.getContainerClient("tokens");
    await container.createIfNotExists();

    const blob = container.getBlockBlobClient(body.token + ".json");
    const data = JSON.stringify({
      token:      body.token,
      company:    body.company,
      email:      body.email,
      type:       body.type,
      notes:      body.notes,
      expiresAt:  body.expiresAt,
      status:     "pending",
      createdAt:  new Date().toISOString()
    });
    await blob.upload(data, Buffer.byteLength(data), {
      blobHTTPHeaders: { blobContentType: "application/json" }
    });

    context.res = { body: { success: true }};
  } catch(e) {
    context.res = { status: 500, body: { success: false, error: e.message }};
  }
};

Function 3 of 3

submit-assessment — Saves completed assessment to Blob Storage & marks token as used

File: ae-functions/submit-assessment/index.js
const { BlobServiceClient } = require("@azure/storage-blob");

module.exports = async function(context, req) {
  const payload = req.body;
  if (!payload) {
    context.res = { status: 400, body: { success: false }};
    return;
  }

  try {
    const client = BlobServiceClient.fromConnectionString(process.env.STORAGE_CONNECTION);
    
    // 1. Save submission to /submissions container
    const submissionsContainer = client.getContainerClient("submissions");
    await submissionsContainer.createIfNotExists();
    
    const ts = new Date().toISOString().replace(/[:.]/g, "-");
    const company = (payload.clientCompany || "unknown").replace(/\s+/g, "-").slice(0, 30);
    const filename = `${ts}_${company}_${payload.type || "assessment"}.json`;
    
    const submissionBlob = submissionsContainer.getBlockBlobClient(filename);
    const data = JSON.stringify(payload, null, 2);
    await submissionBlob.upload(data, Buffer.byteLength(data), {
      blobHTTPHeaders: { blobContentType: "application/json" }
    });

    // 2. Mark token as submitted (if token present)
    if (payload.token) {
      const tokensContainer = client.getContainerClient("tokens");
      const tokenBlob = tokensContainer.getBlockBlobClient(payload.token + ".json");
      try {
        const dl = await tokenBlob.download();
        const txt = await streamToText(dl.readableStreamBody);
        const tokenData = JSON.parse(txt);
        tokenData.status = "submitted";
        tokenData.submittedAt = new Date().toISOString();
        const updated = JSON.stringify(tokenData);
        await tokenBlob.upload(updated, Buffer.byteLength(updated), {
          blobHTTPHeaders: { blobContentType: "application/json" }
        });
      } catch(e) { /* token not found — continue anyway */ }
    }

    context.res = { body: { success: true, filename }};
  } catch(e) {
    context.res = { status: 500, body: { success: false, error: e.message }};
  }
};

async function streamToText(stream) {
  const chunks = [];
  for await (const chunk of stream) chunks.push(chunk);
  return Buffer.concat(chunks).toString("utf-8");
}

Step 6 — List Submissions API

Optional: Enable Cloud-Synced Admin Submissions View

This optional function lets the admin portal fetch submissions directly from Azure Blob Storage, so assessments submitted from any browser appear in the admin portal — not just the local browser that submitted them.

Function 4 (Optional)

list-submissions — Returns all stored assessment JSON files

File: ae-functions/list-submissions/index.js
const { BlobServiceClient } = require("@azure/storage-blob");

module.exports = async function(context, req) {
  // Simple admin key check — set AE_ADMIN_KEY in Function App settings
  const adminKey = req.headers["x-admin-key"];
  if (adminKey !== process.env.AE_ADMIN_KEY) {
    context.res = { status: 401, body: { error: "Unauthorised" }};
    return;
  }

  try {
    const client = BlobServiceClient.fromConnectionString(process.env.STORAGE_CONNECTION);
    const container = client.getContainerClient("submissions");
    
    const submissions = [];
    for await (const blob of container.listBlobsFlat()) {
      const blobClient = container.getBlobClient(blob.name);
      const dl = await blobClient.download();
      const txt = await streamToText(dl.readableStreamBody);
      try { submissions.push(JSON.parse(txt)); } catch(e) {}
    }
    
    submissions.sort((a,b) =>
      new Date(b.submittedAt || b.timestamp) - new Date(a.submittedAt || a.timestamp)
    );
    
    context.res = { body: { submissions }};
  } catch(e) {
    context.res = { status: 500, body: { error: e.message }};
  }
};

async function streamToText(stream) {
  const chunks = [];
  for await (const chunk of stream) chunks.push(chunk);
  return Buffer.concat(chunks).toString("utf-8");
}
